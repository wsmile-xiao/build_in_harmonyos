cmake_minimum_required(VERSION 3.15)
project(test_plutosvg C)
find_package(plutosvg REQUIRED)
find_package(plutovg REQUIRED)
find_path(PLUTOVG_INCLUDE_DIR plutovg.h PATH_SUFFIXES plutovg)
add_executable(test_plutosvg test.c)
target_include_directories(test_plutosvg PRIVATE ${PLUTOVG_INCLUDE_DIR})
target_link_libraries(test_plutosvg PRIVATE plutosvg::plutosvg plutovg::plutovg)
enable_testing()
add_test(NAME test_plutosvg COMMAND test_plutosvg)
