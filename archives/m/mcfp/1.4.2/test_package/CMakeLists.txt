cmake_minimum_required(VERSION 3.16)
project(test_mcfp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
find_package(mcfp REQUIRED CONFIG)
add_executable(test_mcfp test.cpp)
target_link_libraries(test_mcfp mcfp::mcfp)
add_test(NAME test_mcfp COMMAND test_mcfp)
