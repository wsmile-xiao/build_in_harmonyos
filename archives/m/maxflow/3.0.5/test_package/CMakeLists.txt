cmake_minimum_required(VERSION 3.16)
project(test_maxflow LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
find_package(maxflow REQUIRED CONFIG)
add_executable(test_maxflow test_maxflow.cpp)
target_link_libraries(test_maxflow maxflow::maxflow)
add_test(NAME test_maxflow COMMAND test_maxflow)
