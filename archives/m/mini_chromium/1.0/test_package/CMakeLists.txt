cmake_minimum_required(VERSION 3.16)
project(test_mini_chromium CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(mini_chromium REQUIRED)
add_executable(test_mini_chromium test.cpp)
target_link_libraries(test_mini_chromium PRIVATE mini_chromium::mini_chromium)
enable_testing()
add_test(NAME mini_chromium_test COMMAND test_mini_chromium)
