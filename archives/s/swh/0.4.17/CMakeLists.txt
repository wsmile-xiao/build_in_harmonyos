cmake_minimum_required(VERSION 3.20)
project(swh_plugins VERSION 0.4.17 LANGUAGES C)
find_package(fftw3f CONFIG REQUIRED)
include(CheckSymbolExists)
include(TestBigEndian)
set(CMAKE_REQUIRED_LIBRARIES m)
check_symbol_exists(lrintf "math.h" HAVE_LRINTF)
test_big_endian(WORDS_BIGENDIAN)
configure_file(config.h.cmake "${CMAKE_CURRENT_SOURCE_DIR}/config.h")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_STANDARD 99)
add_compile_options(-fno-strict-aliasing)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}")
foreach(util rms db blo pitchscale iir)
  add_library(swh_${util} STATIC util/${util}.c)
  target_link_libraries(swh_${util} PRIVATE m fftw3f::fftw3f)
endforeach()
add_library(swh_gsm STATIC gsm/add.c gsm/decode.c gsm/gsm_decode.c gsm/gsm_encode.c
  gsm/long_term.c gsm/preprocess.c gsm/short_term.c gsm/code.c gsm/gsm_create.c
  gsm/gsm_destroy.c gsm/gsm_option.c gsm/lpc.c gsm/rpe.c gsm/table.c)
add_library(swh_gverb STATIC gverb/gverb.c gverb/gverbdsp.c)
include(generated/plugins.cmake)
list(LENGTH SWH_PLUGINS plugin_count)
if(NOT plugin_count EQUAL 96)
  message(FATAL_ERROR "Upstream module inventory changed")
endif()
foreach(plugin IN LISTS SWH_PLUGINS)
  add_library(${plugin} MODULE generated/${plugin}.c)
  set_target_properties(${plugin} PROPERTIES PREFIX "")
  target_link_libraries(${plugin} PRIVATE m)
  if(plugin MATCHES "^pitch_scale_")
    target_link_libraries(${plugin} PRIVATE swh_pitchscale fftw3f::fftw3f)
  elseif(plugin MATCHES "^(mbeq|imp)_")
    target_link_libraries(${plugin} PRIVATE fftw3f::fftw3f)
  elseif(plugin MATCHES "^(hermes_filter|fm_osc|analogue_osc)_")
    target_link_libraries(${plugin} PRIVATE swh_blo)
  elseif(plugin MATCHES "^(sc1|sc2|sc3|sc4|sc4m|se4)_")
    target_link_libraries(${plugin} PRIVATE swh_db swh_rms)
  elseif(plugin MATCHES "^gsm_")
    target_link_libraries(${plugin} PRIVATE swh_gsm)
  elseif(plugin MATCHES "^gverb_")
    target_link_libraries(${plugin} PRIVATE swh_gverb)
  elseif(plugin MATCHES "^(highpass_iir|lowpass_iir|bandpass_iir|bandpass_a_iir|notch_iir|butterworth)_")
    target_link_libraries(${plugin} PRIVATE swh_iir)
  endif()
  install(TARGETS ${plugin} LIBRARY DESTINATION lib/ladspa)
endforeach()
install(FILES ladspa.h DESTINATION include)
