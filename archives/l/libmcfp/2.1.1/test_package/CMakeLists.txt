cmake_minimum_required(VERSION 3.16)
project(test_libmcfp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
find_package(libmcfp REQUIRED CONFIG)
add_executable(test_libmcfp test.cpp)
target_link_libraries(test_libmcfp libmcfp::libmcfp)
add_test(NAME test_libmcfp COMMAND test_libmcfp)
