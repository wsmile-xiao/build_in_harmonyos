cmake_minimum_required(VERSION 3.15)
project(test_liquidsfz LANGUAGES CXX)
find_package(liquidsfz CONFIG REQUIRED)
add_executable(test_liquidsfz src/test_liquidsfz.cpp)
target_compile_features(test_liquidsfz PRIVATE cxx_std_17)
target_link_libraries(test_liquidsfz PRIVATE liquidsfz::liquidsfz)
if(CMAKE_SYSTEM_NAME STREQUAL "HarmonyOS" OR CMAKE_SYSTEM_NAME STREQUAL "OHOS")
    add_executable(test_liquidsfz_ohos src/test_liquidsfz_ohos.cpp)
    target_compile_features(test_liquidsfz_ohos PRIVATE cxx_std_17)
    target_link_libraries(test_liquidsfz_ohos PRIVATE liquidsfz::liquidsfz)
endif()
