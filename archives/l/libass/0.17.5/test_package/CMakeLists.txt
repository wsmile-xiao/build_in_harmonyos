cmake_minimum_required(VERSION 3.15)
project(test_libass C)
find_package(libass CONFIG REQUIRED)
add_executable(test_libass test.c)
target_link_libraries(test_libass PRIVATE libass::ass)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
    target_link_options(test_libass PRIVATE -stdlib=libc++)
endif()
enable_testing()
add_test(NAME test_libass COMMAND $<TARGET_FILE:test_libass>)
