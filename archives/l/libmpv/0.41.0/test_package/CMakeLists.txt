cmake_minimum_required(VERSION 3.15)
project(test_libmpv C)
find_package(libmpv CONFIG REQUIRED)
add_executable(test_libmpv test.c)
target_link_libraries(test_libmpv PRIVATE mpv::mpv)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
    target_link_options(test_libmpv PRIVATE -stdlib=libc++)
endif()
enable_testing()
add_test(NAME test_libmpv COMMAND $<TARGET_FILE:test_libmpv>)
