cmake_minimum_required(VERSION 3.21)
project(libwebp_actual_consumer C)
find_package(WebP REQUIRED CONFIG)
add_executable(consumer consumer.c)
target_link_libraries(consumer PRIVATE WebP::webp WebP::webpmux WebP::webpdemux)
set_target_properties(consumer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(UNIX OR CMAKE_SYSTEM_NAME STREQUAL "OHOS" OR CMAKE_SYSTEM_NAME STREQUAL "HarmonyOS")
  add_executable(harmony_native_loader harmony_native_loader.c)
  target_link_libraries(harmony_native_loader PRIVATE ${CMAKE_DL_LIBS})
  set_target_properties(harmony_native_loader PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
