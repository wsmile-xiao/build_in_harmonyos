cmake_minimum_required(VERSION 3.15)
project(test_libesmtp C)
find_package(libesmtp CONFIG REQUIRED)
add_executable(test_libesmtp test.c)
target_link_libraries(test_libesmtp PRIVATE libesmtp::libesmtp)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
    target_link_options(test_libesmtp PRIVATE -stdlib=libc++)
endif()
enable_testing()
add_test(NAME test_libesmtp COMMAND $<TARGET_FILE:test_libesmtp>)
