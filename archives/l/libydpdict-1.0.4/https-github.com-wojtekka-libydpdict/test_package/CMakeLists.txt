cmake_minimum_required(VERSION 3.15)
project(test_libydpdict C)
find_package(libydpdict REQUIRED)
enable_testing()

foreach(test IN ITEMS test_libydpdict test_ydpdict_e2e)
    if(test STREQUAL "test_libydpdict")
        set(source test.c)
    else()
        set(source test_ydpdict_e2e.c)
    endif()
    add_executable(${test} ${source})
    target_link_libraries(${test} PRIVATE libydpdict::libydpdict)
    target_compile_features(${test} PRIVATE c_std_99)
    add_test(NAME ${test} COMMAND ${test})
endforeach()

option(YDPDICT_HARMONY_TEST "Run the HarmonyOS user storage workflow" OFF)
if(YDPDICT_HARMONY_TEST)
    add_executable(test_harmony_storage test_harmony_storage.c)
    target_link_libraries(test_harmony_storage PRIVATE libydpdict::libydpdict)
    target_compile_features(test_harmony_storage PRIVATE c_std_99)
    add_test(NAME test_harmony_storage COMMAND test_harmony_storage)
endif()
