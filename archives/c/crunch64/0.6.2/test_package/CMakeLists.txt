cmake_minimum_required(VERSION 3.15)
project(crunch64_consumer C)
find_package(crunch64 CONFIG REQUIRED)
enable_testing()
add_executable(consumer consumer.c)
target_link_libraries(consumer PRIVATE crunch64::crunch64)
target_compile_features(consumer PRIVATE c_std_11)
target_compile_options(consumer PRIVATE -Wall -Wextra -Werror)
foreach(test yaz0 yay0 mio0 errors gzip)
    add_test(NAME consumer_${test} COMMAND consumer ${test})
endforeach()
if(CMAKE_SYSTEM_NAME MATCHES "OHOS|HarmonyOS")
    add_executable(harmonyos_test harmonyos_test.c)
    target_compile_features(harmonyos_test PRIVATE c_std_11)
    target_compile_options(harmonyos_test PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(harmonyos_test PRIVATE crunch64::crunch64 deviceinfo_ndk.z)
    add_test(NAME harmonyos_deviceinfo_mmap COMMAND harmonyos_test)
endif()
