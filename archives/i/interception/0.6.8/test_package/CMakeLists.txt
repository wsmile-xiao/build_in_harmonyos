cmake_minimum_required(VERSION 3.15)
project(interception_consumer LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBEVDEV REQUIRED IMPORTED_TARGET libevdev)
pkg_check_modules(YAMLCPP REQUIRED IMPORTED_TARGET yaml-cpp)
configure_file(production_source.h.in production_source.h @ONLY)
add_executable(interception_config_probe config_probe.cpp)
target_compile_features(interception_config_probe PRIVATE cxx_std_11)
target_include_directories(interception_config_probe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(interception_config_probe PRIVATE PkgConfig::LIBEVDEV PkgConfig::YAMLCPP)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  target_link_options(interception_config_probe PRIVATE "-Wl,--disable-new-dtags")
endif()
